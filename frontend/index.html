Python (Direct Service Import) 
from app.services.inference import InferenceService

# High-speed chat completion with 6-key auto-failover
response = await InferenceService.chat_completion(
    model="qwen/qwen3.8-27b",
    messages=[
        {"role": "system", "content": "You are an expert coder."},
        {"role": "user", "content": "Write a debounce function in JS."}
    ],
    temperature=0.3
)

print(response["choices"][0]["message"]["content"])
print(f"Served by: {response['_rotator']['key_name']} in {response['_rotator']['latency_ms']}ms")
OpenAI Python SDK (Local Proxy) 
from openai import OpenAI

# Zero-configuration: points to your running local proxy!
client = OpenAI(
    base_url="http://localhost:8000/v1",
    api_key="any-string" # Proxy automatically rotates the 6 keys
)

completion = client.chat.completions.create(
    model="openai/gpt-oss-120b",
    messages=[{"role": "user", "content": "Explain async await"}],
    temperature=0.7
)

print(completion.choices[0].message.content)
