from app.services.inference import InferenceService

# High-speed chat completion with automatic 6-key rotation and failover

        model="qwen/qwen3.8-27b",
        messages=[
            {"role": "system", "content": "You are an expert AI assistant."},

⚡ Python Real-Time SSE Token Streaming 
from app.services.inference import InferenceService

# Real-time token streaming with failover on initial connection
async def stream_response():
    messages = [{"role": "user", "content": "أهلاً بك، ما هي أهم مميزات الذكاء الاصطناعي؟"}]
    
    async for chunk in InferenceService.stream_chat(
        model="allam-2-7b",
        messages=messages,
        temperature=0.7
    ):
        if chunk.get("type") == "token":
            print(chunk["token"], end="", flush=True)
        elif chunk.get("type") == "done":
            meta = chunk["meta"]
            print(f"\n\n[Done: {meta['tokens']} tokens | {meta['latency_ms']}ms | {meta['key_name']}]")
